-- 1. CREACIÓN DE TABLAS

create table alternativa (
    es_correcta boolean not null,
    letra char(1) not null check ((letra in ('A','B','C','D','E'))),
    orden_visualizacion integer,
    id bigint generated by default as identity,
    pregunta_id bigint not null,
    tipo varchar(20) not null check ((tipo in ('TEXTO','IMAGEN'))),
    imagen_url varchar(1000),
    contenido_texto TEXT,
    primary key (id)
);

create table categoria_curso_config (
    activo boolean not null,
    cantidad_sugerida integer,
    categoria_examen_id bigint not null,
    curso_id bigint not null,
    fecha_configuracion timestamp(6) not null,
    id bigint generated by default as identity,
    primary key (id),
    constraint uk_categoria_curso_config unique (categoria_examen_id, curso_id)
);

create table categoria_examen (
    activo boolean not null,
    fecha_creacion timestamp(6) not null,
    id bigint generated by default as identity,
    nombre varchar(120) not null,
    descripcion varchar(500),
    primary key (id)
);

create table curso (
    activo boolean not null,
    fecha_creacion timestamp(6) not null,
    id bigint generated by default as identity,
    codigo varchar(20),
    nombre varchar(120) not null,
    descripcion varchar(500),
    primary key (id)
);

create table examen (
    aleatorizar_alternativas boolean not null,
    aleatorizar_preguntas boolean not null,
    cantidad_versiones integer not null,
    duracion_minutos integer not null,
    categoria_examen_id bigint not null,
    fecha_creacion timestamp(6) not null,
    fecha_publicacion timestamp(6),
    id bigint generated by default as identity,
    color_portada varchar(20),
    estado varchar(30) not null check ((estado in ('BORRADOR','ACTIVO','FINALIZADO','ANULADO'))),
    codigo varchar(50) not null unique,
    modalidad varchar(150),
    nombre varchar(150) not null,
    nombre_universidad varchar(150),
    titulo_examen varchar(150),
    descripcion varchar(500),
    logo_url TEXT,
    primary key (id)
);

create table examen_configuracion_curso (
    cantidad_dificil integer,
    cantidad_facil integer,
    cantidad_medio integer,
    cantidad_preguntas integer not null,
    puntaje_por_pregunta float(53) not null,
    curso_id bigint not null,
    examen_id bigint not null,
    id bigint generated by default as identity,
    primary key (id)
);

create table examen_pregunta (
    numero_orden integer not null,
    orden integer not null,
    puntaje float(53) not null,
    examen_version_id bigint not null,
    id bigint generated by default as identity,
    pregunta_id bigint not null,
    alternativas_ordenadas TEXT not null,
    primary key (id)
);

create table examen_version (
    numero integer not null,
    examen_id bigint not null,
    fecha_generacion timestamp(6) not null,
    id bigint generated by default as identity,
    codigo_version varchar(50) not null,
    primary key (id)
);

create table historial_accion (
    examen_id bigint,
    fecha_accion timestamp(6) not null,
    id bigint generated by default as identity,
    tipo_accion varchar(40) not null check ((tipo_accion in ('CREACION','ACTUALIZACION','ELIMINACION','GENERAR_EXAMEN','GENERACION_EXAMEN','PUBLICACION_EXAMEN','ANULACION_EXAMEN'))),
    modulo varchar(80) not null,
    usuario_responsable varchar(120) not null,
    descripcion TEXT not null,
    primary key (id)
);

create table pregunta (
    activo boolean not null,
    curso_id bigint not null,
    fecha_creacion timestamp(6) not null,
    id bigint generated by default as identity,
    codigo varchar(30) not null unique,
    dificultad varchar(30) not null check ((dificultad in ('FACIL','MEDIO','DIFICIL'))),
    imagen_url varchar(1000),
    enunciado TEXT not null,
    primary key (id)
);
